\begin{abstract}
עבודה זו בוחנת אם מודל מרקוב חבוי גאוסיאני (\tech{Gaussian Hidden Markov Model; HMM}) הלומד משטרים סטטיסטיים סמויים משפר חיזוי של סימן התשואה ביום הבא ביחס למודלי בסיס פשוטים. הניסוי הקנוני כולל חמישה נכסים שנקבעו מראש — \asset{SPY}, \asset{GLD}, \asset{TLT}, \asset{BTC-USD} ו־\asset{DIS} — ונתונים יומיים שהתבקשו לטווח \datecode{2014-01-01--2024-12-31}. וקטור התכונות כולל תשואה לוגריתמית, תנודתיות נגררת ל־20 ימים, טווח תוך־יומי ושינוי לוגריתמי בנפח. החלוקה כרונולוגית וקבועה: כ־\pct{70} אימון, \pct{15} אימות ו־\pct{15} מבחן מוחזק־בצד. זהו \textbf{פיצול מוחזק־בצד יחיד לכל נכס, ולא ניסוי \tech{rolling-origin}}. שורות הגבול הוסרו כך שמטרת היום הבא בכל מקטע אינה חוצה למקטע הבא; ההתאמה של הסקיילר נעשתה על האימון בלבד.

לכל נכס הותאמו מודלים עם $K\in\{2,3,4\}$ מצבים ושלושה זרעי אתחול קבועים. התצורה נבחרה לפי \tech{validation log-likelihood} בלבד, ולא לפי ביצועי המבחן. כל 45 ההתאמות התכנסו. במבחן, דיוק הכיוון של ה־\tech{HMM} הנבחר היה \pct{58.50}, \pct{53.40}, \pct{50.24}, \pct{50.45} ו־\pct{51.46}, בהתאמה לסדר הנכסים לעיל. ממוצע בלתי־משוקלל בין הנכסים היה \pct{52.81}. מודל ממוצע האימון הפשוט השיג \pct{52.99}: ארבע תוצאות היו שוויון וב־\asset{BTC-USD} ה־\tech{HMM} היה נמוך ב־0.89 נקודת אחוז. לכן הממצא המרכזי הוא שלילי וזהיר: \textbf{בניסוי הקנוני ה־\tech{HMM} לא גבר על מודל ממוצע האימון במדד \tech{DPA}}. לצד זאת, המצבים שנלמדו הציגו הבדלים ניכרים בתנודתיות ובהתמדה, ולכן יש למודל ערך תיאורי אפשרי כייצוג של משטרים סטטיסטיים. אין בעבודה טענה לרווחיות מסחר, למובהקות סטטיסטית או לעליונות כללית של המודל.
\end{abstract}

\textbf{מילות מפתח:} \tech{Hidden Markov Model}, משטרי שוק, סדרות זמן פיננסיות, חיזוי כיוון, זליגת מידע, \tech{DPA}, בחירת מודל.

\section{מבוא ומוטיבציה}
סדרות זמן פיננסיות אינן מתנהגות בהכרח כתהליך יחיד ויציב. תקופות שקטות יחסית מתחלפות בתקופות תנודתיות, מבנה התלות עשוי להשתנות, ותצפיות קיצוניות נוטות להופיע באשכולות. רעיון ``משטר השוק'' מנסח אפשרות שלפיה ההתפלגות היוצרת את התצפיות משתנה לאורך הזמן בהתאם למצב שאינו נצפה ישירות. מודל מרקוב חבוי נותן לרעיון זה ייצוג הסתברותי מפורש: שרשרת מרקוב בדידה מתארת את המעברים בין המצבים הסמויים, והתפלגות פליטה מתארת את התצפיות בתנאי המצב \cite{rabiner1989tutorial,murphy2012machine}. שימוש במשטרים סמויים בניתוח כלכלי ופיננסי מוכר גם ממודלי החלפת משטרים \cite{hamilton1989new}.

המוטיבציה אינה להניח מראש שקיימים ``שוק שורי'' ו``שוק דובי'', אלא לבדוק אם חלוקה סטטיסטית שנלמדת מן הנתונים: (א) יוצרת מצבים בעלי מאפיינים נבדלים; ו־(ב) מסייעת לחיזוי כיוון התשואה ביום הבא ביחס לכללים פשוטים. ההבחנה חשובה: מודל יכול לתאר היטב אשכולות תנודתיות בלי לשפר תחזית נקודתית, ולהפך. כמו כן, הצלחה במדד כיוון אינה שקולה לרווח לאחר עלויות עסקה, מגבלות ביצוע וסיכון.

העבודה נשענת רעיונית על הצעת הפרויקט ועל מאמרם של \textenglish{Catello et al.} \cite{catello2023hmm}, אך אינה משחזרת את המימוש שלהם. המאמר משתמש בייצוג ובפרוטוקול משלו, ואילו כאן נבחן \tech{Gaussian HMM} רב־ממדי, מספר מצבים $K=2,3,4$, חמישה נכסים ומודלי בסיס אחידים. שינוי מרכזי ביחס לחלק מהניסויים ההיסטוריים במאגר הוא פרוטוקול קנוני המתקן זליגת מטרות בגבולות ומונע בחירה ישירה על המבחן.

\subsection{תרומות העבודה}
\begin{enumerate}
  \item ניסוח השוואה מבוקרת בין \tech{Gaussian HMM} לבין ארבעה מודלי בסיס תחת אותו אופק חיזוי ואותו מבחן כרונולוגי.
  \item שימוש בחלוקת \tech{Train/Validation/Test} מוגנת־מטרה, ובהתאמת \tech{StandardScaler} על האימון בלבד.
  \item בדיקת $K\in\{2,3,4\}$ ושלושה אתחולים לכל $K$, תוך תיעוד התכנסות, מספר איטרציות, \tech{AIC}, \tech{BIC} וזמני ריצה.
  \item הפרדה מפורשת בין מדדי חיזוי לבין פרשנות פנימית של מצבים, מטריצות מעבר והתמדה.
  \item דיווח מלא של תוצאה שלילית: אין יתרון \tech{DPA} ל־\tech{HMM} על פני ממוצע האימון בניסוי הקנוני.
\end{enumerate}

\section{שאלות המחקר והציפיות האמפיריות}
שאלת המחקר הראשית, בהתאם להצעה המקורית, היא:

\begin{quote}
\textbf{האם \tech{HMM} הלומד מצבי שוק סמויים משפר את חיזוי כיוון השינוי היומי ביחס למודלים פשוטים, וכיצד מספר המצבים החבויים משפיע על התוצאה?}
\end{quote}

מן השאלה הראשית נגזרות ארבע שאלות אופרטיביות:
\begin{enumerate}
  \item \textbf{ביצועי חיזוי:} האם ה־\tech{DPA} של המודל שנבחר באימות גבוה מזה של מודל ממוצע האימון, התמדה, ממוצע נע ושרשרת מרקוב נצפית?
  \item \textbf{מורכבות ובחירה:} איזה שילוב של $K$ וזרע משיג את \tech{validation log-likelihood} הגבוה ביותר, והאם ההתאמות מתכנסות?
  \Needspace{4\baselineskip}
  \item \textbf{מבנה סמוי:} האם המצבים שונים בתשואה הממוצעת, בתנודתיות, בשכיחות ובמשך הרצף הממוצע?
  \item \textbf{הכללה בין נכסים:} האם דפוס הביצועים עקבי בנכס מנייתי רחב, זהב, אג"ח ארוך, קריפטו ומניה בודדת?
\end{enumerate}

הציפייה התאורטית אינה מנוסחת כהשערת עליונות קשיחה. ל־\tech{HMM} יתרון ייצוגי אפשרי כאשר יש מצבים מתמידים בעלי התפלגויות מובחנות; אך תשואות יומיות רועשות מאוד, ובסיס פשוט החוזה תמיד את סימן ממוצע האימון עשוי להיות קשה לניצוח כאשר שיעור הימים החיוביים גבוה מ־\pct{50}. לפיכך התוצאה השלילית היא תוצאה מדעית לגיטימית ולא כשל של הדיווח.

\section{רקע תאורטי: מודל מרקוב חבוי גאוסיאני}
\subsection{הגדרת המודל}
נסמן ב־$S_t\in\{1,\ldots,K\}$ את המצב הסמוי ביום $t$, וב־$\mathbf{x}_t\in\mathbb{R}^{d}$ את וקטור התכונות הנצפה. הנחת מרקוב מסדר ראשון היא
\begin{equation}
P(S_t=j\mid S_{t-1}=i,S_{t-2},\ldots,S_1)
=P(S_t=j\mid S_{t-1}=i)=a_{ij},
\end{equation}
כאשר $A=(a_{ij})$ היא מטריצת המעבר וכל שורה בה מסתכמת ל־1. וקטור ההתחלה הוא $\boldsymbol{\pi}$, עם $\pi_i=P(S_1=i)$.

ב־\tech{Gaussian HMM} רב־ממדי, התפלגות הפליטה היא
\begin{equation}
\mathbf{x}_t\mid S_t=k \sim \mathcal{N}(\boldsymbol{\mu}_k,\boldsymbol{\Sigma}_k).
\end{equation}
בניסוי זה $d=4$ וסוג הקו־וריאנציה הוא \tech{full}; לכן כל מצב מקבל מטריצת קו־וריאנציה מלאה משלו. הלייקלי-הוד של רצף התצפיות מתקבל מסכימה על כל רצפי המצבים האפשריים:
\begin{equation}
P(\mathbf{x}_{1:T}\mid\theta)=
\sum_{s_{1:T}} \pi_{s_1}
\left[\prod_{t=2}^{T}a_{s_{t-1},s_t}\right]
\left[\prod_{t=1}^{T}p(\mathbf{x}_t\mid S_t=s_t)\right].
\end{equation}
חישוב יעיל נעשה באמצעות אלגוריתם \tech{Forward}; הפרמטרים נאמדים באלגוריתם \tech{Baum--Welch}, מקרה של \tech{Expectation--Maximization} \cite{baum1970maximization,rabiner1989tutorial}.

\subsection{מספר פרמטרים ומורכבות}
עבור $K$ מצבים, $d$ תכונות וקו־וריאנציה מלאה, מספר הפרמטרים החופשיים ששימש בארטיפקטים הוא
\begin{equation}
p=(K-1)+K(K-1)+Kd+K\frac{d(d+1)}{2}.
\end{equation}
כאשר $d=4$, מתקבלים בהתאמה $p=31,50,71$ עבור $K=2,3,4$. קריטריוני המידע חושבו על האימון:
\begin{align}
\mathrm{AIC} &= 2p-2\ell_{\mathrm{train}},\\
\mathrm{BIC} &= p\log(n_{\mathrm{train}})-2\ell_{\mathrm{train}}.
\end{align}
ערך נמוך יותר עדיף בהינתן אותו סט נתונים, אך בפרוטוקול הקנוני הם נשמרו כאבחונים בלבד; הבחירה בפועל נעשתה לפי לייקלי-הוד באימות. העיקרון של איזון התאמה ומורכבות מוכר מ־\cite{akaike1974new,schwarz1978estimating}; בחירת מספר מצבים ב־\tech{HMM} אינה רק שאלה של ערך קריטריון יחיד, אלא גם של יציבות ופרשנות \cite{pohle2017selecting}.

\subsection{פענוח ותחזית}
מצבי המבחן פוענחו באופן ``עבר בלבד'': עבור כל יום מבחן $t$ נבנה רצף המורכב לכל היותר מ־100 תצפיות האימון האחרונות ומכל תצפיות המבחן עד $t$; המודל מפענח את הרצף והמצב האחרון נלקח כ־$\hat S_t$. לכן תצפיות מבחן מ־$t+1$ ואילך אינן משמשות לפענוח יום $t$. חשוב לדייק: זהו \tech{Viterbi-prefix decoding}, לא הסתברות מסוננת מלאה ולא החלקה דו־כיוונית.

לכל מצב חושב ממוצע תשואת האימון $\bar r_k$. תחזית התשואה ליום הבא היא
\begin{equation}
\widehat r_{t+1}=\sum_{j=1}^{K}a_{\hat S_t,j}\bar r_j.
\end{equation}
כלומר, התחזית משלבת בין הערכת המצב הנוכחי, הסתברויות המעבר וממוצעי התשואה של המצבים. התווית החזויה היא חיובית אם $\widehat r_{t+1}\geq0$ ושלילית אחרת. אותו סף, $\geq0$, משמש לכל המודלים.

\section{נתונים, נכסים ובניית תכונות}
\subsection{יקום הנכסים ומקור הנתונים}
היקום הוגדר מראש ואינו תוצאה של חיפוש נכסים שבהם המודל הצליח:
\begin{itemize}
  \item \asset{SPY} — קרן סל רחבה על שוק המניות האמריקאי;
  \item \asset{GLD} — חשיפה לזהב;
  \item \asset{TLT} — אג"ח ממשלת ארה"ב לטווח ארוך;
  \item \asset{BTC-USD} — נכס קריפטוגרפי הנסחר גם בסופי שבוע;
  \item \asset{DIS} — מניה בודדת, הנשמרת גם כמקרה מחקר היסטורי אך אינה ``מנצחת'' שנבחרה לפי המבחן.
\end{itemize}

הנתונים נטענו באמצעות \tech{yfinance} \cite{yfinance} כנתוני \tech{OHLCV} יומיים, עם תאריך התחלה מבוקש \datecode{2014-01-01} ותאריך סיום מבוקש \datecode{2024-12-31}. לחישוב תשואה ומחיר יעד נעשה שימוש ב־\tech{Adjusted Close} כאשר הוא זמין; הטווח התוך־יומי משתמש ב־\tech{High}, \tech{Low} ו־\tech{Close}. רשומות חסרות או אינסופיות הוסרו. מאחר שלוחות המסחר שונים, אין להסיק שמספר התצפיות זהה בין \asset{BTC-USD} לבין הנכסים הסחירים בבורסה.

ה־\tech{manifest} הקנוני שומר את טווח הבקשה ואת שיעורי החלוקה, אך אינו שומר את מספר התצפיות ואת תאריכי הגבול בפועל לכל נכס. מסיבה זו איננו ממציאים כאן תאריכים מדויקים שאינם קיימים בארטיפקט. חוסר זה מופיע במפורש במגבלות ובהמלצות לשחזור עתידי.

\subsection{וקטור התכונות}
עבור מחיר מתואם $P_t$, נפח $V_t$, גבוה $H_t$, נמוך $L_t$ וסגירה $C_t$, נבנו ארבע תכונות:
\begin{align}
r_t &= \log(P_t/P_{t-1}),\\
\sigma^{(20)}_t &= \operatorname{sd}(r_{t-19},\ldots,r_t),\\
q_t &= (H_t-L_t)/C_t,\\
v_t &= \log(V_t/V_{t-1}).
\end{align}
מכאן $\mathbf{x}_t=(r_t,\sigma^{(20)}_t,q_t,v_t)$. המטרה היא $y_t=r_{t+1}$, ותאריך המטרה נשמר בנפרד כדי לאפשר בדיקת גבולות. התכונות משתמשות במידע עד סוף יום $t$ בלבד. לכן תזמון התחזית המוצהר הוא \textbf{לאחר סגירת יום $t$}, כאשר הגבוה, הנמוך והנפח של אותו יום כבר ידועים, לצורך תחזית יום $t+1$.

בחירת התכונות מנסה לתאר ארבעה היבטים: כיוון ועוצמת התנועה האחרונה; תנודתיות מקומית; טווח תוך־יומי; ושינוי בפעילות המסחר. היא אינה כוללת אינדיקטורים מקרו־כלכליים, חדשות, ספר פקודות או נתונים עתידיים. חלון התנודתיות גורם להסרת תצפיות הפתיחה שאין להן 20 ימי היסטוריה מלאה.

\subsection{נרמול}
לכל נכס הותאם \tech{StandardScaler} על מטריצת האימון בלבד:
\begin{equation}
z_{t,j}=\frac{x_{t,j}-\widehat\mu_{j,\mathrm{train}}}
{\widehat\sigma_{j,\mathrm{train}}}.
\end{equation}
אותם פרמטרים שימשו להתמרת האימות והמבחן. שלב זה חיוני משום שהפליטה הגאוסיאנית רגישה לקני מידה, והתאמת הסקיילר על כל התקופה הייתה מכניסה מידע מן העתיד.

\section{מודלי הבסיס}
השוואה למודל מורכב היא משמעותית רק כאשר הבסיסים פשוטים, שקופים ומתוזמנים לאותו יעד.

\subsection{ממוצע אימון}
מודל \tech{Naive -- train mean} חוזה לכל יום את ממוצע תשואת היום הבא באימון:
\begin{equation}
\widehat r_{t+1}=\frac{1}{n_{\mathrm{train}}}\sum_{i\in\mathrm{train}}r_{i+1}.
\end{equation}
זהו בסיס חשוב במיוחד ל־\tech{DPA}: אם ממוצע האימון חיובי, כל התחזיות מסווגות כחיוביות, ולכן הדיוק שווה למעשה לשיעור הימים הלא־שליליים במבחן. הוא בוחן אם המבנה המורכב של ה־\tech{HMM} מספק מידע כיווני מעבר לנטייה השולית.

\subsection{התמדה}
מודל \tech{Naive -- persistence} קובע $\widehat r_{t+1}=r_t$. הוא מייצג השערה קצרה ופשוטה שהשינוי האחרון ממשיך ליום הבא. בשוק שבו סימני תשואה כמעט בלתי־תלויים, ביצועיו צפויים להיות קרובים לאקראי.

\subsection{ממוצע נע}
מודל \tech{Moving Average 5} חוזה את ממוצע חמש התשואות האחרונות. הוא מפחית רעש אך עלול לפגר אחרי שינוי. בחישוב המבחן הקנוני נבנה הרצף מן האימון ומהמבחן; ערכי מבחן קודמים זמינים בזמן החיזוי, אך נתוני אימות לא צורפו למסלול החישוב. זהו פרט מימוש שיש לשמר בשחזור ולשפר בפרוטוקול הבא.

\subsection{בסיס כיוון מותנה}
המודל המכונה בארטיפקט \tech{Discrete Markov Chain} ממפה את כיוון התשואה הנוכחית $d_t=\mathbb{1}[r_t\geq0]$ לתשואה הצפויה ביום הבא, הנאמדת באימון:
\begin{equation}
\widehat r_{t+1}=\mathbb{E}_{\mathrm{train}}[r_{t+1}\mid d_t].
\end{equation}
לכן זהו למעשה \textbf{מנבא ממוצע מותנה בכיוון} ולא מימוש מלא הנשמר כמטריצת הסתברויות מעבר בין תוויות. השם נשמר לצורך התאמה ל־\code{results.csv}, אך הפרשנות המדעית משתמשת בהגדרה המדויקת.

\section{מדדי הערכה}
\subsection{דיוק כיוון — \tech{DPA}}
נגדיר $d(r)=\mathbb{1}[r\geq0]$. אז
\begin{equation}
\mathrm{DPA}=\frac{1}{N}\sum_{t=1}^{N}
\mathbb{1}\left[d(\widehat r_{t+1})=d(r_{t+1})\right].
\end{equation}
ערך גבוה יותר פירושו ששיעור גדול יותר של סימני התשואה נחזה נכון. המדד עונה ישירות לשאלת המחקר, אך מתעלם מגודל הטעות, מחוסר איזון בין ימים חיוביים ושליליים ומעלויות מסחר. בפרט, \pct{58} אינו ``תשואה של \pct{58}''.

\subsection{שגיאת תשואה: \tech{MAE} ו־\tech{RMSE}}
\begin{align}
\mathrm{MAE} &= \frac{1}{N}\sum_{t=1}^{N}|r_{t+1}-\widehat r_{t+1}|,\\
\mathrm{RMSE} &= \sqrt{\frac{1}{N}\sum_{t=1}^{N}(r_{t+1}-\widehat r_{t+1})^2}.
\end{align}
\tech{MAE} מודד טעות מוחלטת טיפוסית ביחידות של תשואה לוגריתמית. \tech{RMSE} מעניש חזק יותר טעויות גדולות ולכן רגיש לימים קיצוניים. ערך נמוך יותר עדיף. אף אחד מהמדדים אינו מודד איכות של הסתברות מצב או רווחיות.

\subsection{שגיאת מחיר: \tech{MAPE}}
מחיר החיזוי נגזר מן התשואה: $\widehat P_{t+1}=P_t\exp(\widehat r_{t+1})$. לאחר מכן
\begin{equation}
\mathrm{MAPE}=\frac{100}{N}\sum_{t=1}^{N}
\left|\frac{P_{t+1}-\widehat P_{t+1}}{P_{t+1}}\right|.
\end{equation}
המדד אינטואיטיבי באחוזים, אך מחיר יום המחר קרוב בדרך כלל למחיר היום, ולכן \tech{MAPE} נמוך אינו מעיד לבדו על חיזוי כיוון טוב. באופן כללי ל־\tech{MAPE} מגבלות ידועות, בפרט כאשר המכנה קרוב לאפס \cite{hyndman2006another}; בעבודה זו מחירי הנכסים חיוביים, אך המגבלה הפרשנית נותרת.

\subsection{לייקלי-הוד, קריטריוני מידע וזמן}
\tech{Log-likelihood} מודד את התאמת המודל ההסתברותי לרצף התכונות, לא את שגיאת התשואה החזויה. ערכים פחות שליליים טובים יותר כאשר משווים על אותו רצף. \tech{AIC/BIC} מתקנים את לייקלי-הוד של האימון בגין מספר הפרמטרים. זמני הריצה שנשמרו קצרים מאוד אך אינם מדד חומרה אחיד לחלוטין: בבסיסים הם בעיקר זמן חיזוי/הערכה, וב־\tech{HMM} בעיקר זמן התאמה. לכן הם מוצגים כאבחון הנדסי ולא כהשוואת יעילות פורמלית.

\section{דיון}
\subsection{הממצא המרכזי: ייצוג משטרים ללא יתרון כיווני}
הממצא החשוב ביותר הוא שה־\tech{HMM} לא גבר על בסיס ממוצע האימון ב־\tech{DPA}: 0 ניצחונות, 4 תוצאות שוויון והפסד אחד. זהו מבחן חזק יחסית לטענה שהמצבים מוסיפים מידע כיווני. בארבעת מקרי השוויון, ייתכן שהתחזיות של שני המודלים נבדלו בגודל אך לא בסימן, ולכן ה־\tech{DPA} זהה. השוואות ה־\tech{MAE/RMSE} מראות רק הבדלים זעירים ומעורבים, ללא דפוס של עליונות עקבית.

התוצאה אינה סותרת את קיומם של משטרים בתנודתיות. התכונות כוללות במפורש תנודתיות נגררת וטווח יומי, וה־\tech{HMM} מסוגל להפריד היטב אשכולות של פיזור. אך תחזית סימן $r_{t+1}$ דורשת קשר בין המצב הנוכחי לבין תוחלת התשואה העתידית. אם מצבים נבדלים בעיקר בשונות ולא בתוחלת, או אם תוחלות המצב קטנות ולא יציבות, שיפור בייצוג אינו מוביל בהכרח לשיפור ב־\tech{DPA}.

\subsection{מדוע ממוצע האימון הוא בסיס קשה}
ב־\asset{SPY} ממוצע האימון כנראה חיובי ולכן תחזית קבועה חיובית משיגה \pct{58.50}, שיעור הימים הלא־שליליים במבחן. מודל מורכב חייב לא רק לזהות ימים שליליים, אלא לעשות זאת בדיוק מספיק בלי להפוך ימים חיוביים רבים לשליליים. תוצאה זו ממחישה מדוע יש לדווח על הבסיס ולא להשוות ל־\pct{50} באופן נאיבי. \pct{58.50} ל־\tech{HMM} אינו בהכרח עדות ליכולת תזמון, אם אותה תוצאה מתקבלת מתחזית קבועה.

ב־\asset{TLT}, לעומת זאת, \tech{DPA} של ממוצע האימון קרוב ל־\pct{50}, והממוצע הנע עדיף בכ־0.97 נקודת אחוז. זהו רמז אפשרי לכך שתלות קצרה מקומית עשויה להיות שימושית יותר מן החלוקה למשטרים עבור אותו חלון מבחן, אך פיצול יחיד אינו מספיק כדי לקבוע זאת.

\subsection{בחירת $K$ ולייקלי-הוד}
בחירה לפי \tech{validation log-likelihood} העדיפה $K=4$ בארבעה נכסים. מודל מורכב יכול לשפר לייקלי-הוד משום שהוא מתאר טוב יותר התפלגויות הטרוגניות, במיוחד זנבות ואשכולות תנודתיות. אולם היעד של הבחירה והיעד של המחקר אינם זהים: לייקלי-הוד מתייחס לווקטור התכונות, בעוד \tech{DPA} מתייחס לסימן התשואה הבאה. אי־התאמה זו יכולה להסביר מדוע בחירת $K=4$ לא שיפרה חיזוי כיוון.

\tech{AIC/BIC} שנשמרו אינם בסיס לבחירה הקנונית, ואין להשוות ערכיהם בין נכסים בעלי מספרי תצפיות והתפלגויות שונים. בתוך נכס ניתן להשתמש בהם כאבחון משלים, אך נכון יותר בפרוטוקול עתידי להגדיר מראש אם הבחירה מיועדת לצפיפות, לתחזית או לפשרה ביניהן. בחירה על \tech{DPA} באימות הייתה מתאימה יותר לשאלת הכיוון, אך גם רגישה יותר לרעש ולסף; בחירה לפי \tech{BIC} הייתה שמרנית יותר כלפי מורכבות.

\subsection{פרשנות המצבים}
ב־\asset{SPY}, מצב 3 מתאפיין בשכיחות \pct{4.06}, תשואה יומית ממוצעת \pct{-0.442} ותנודתיות \pct{3.710}; זהו הבדל סטטיסטי ברור מן המצבים האחרים. ב־\asset{TLT}, מצב 1 נדיר ותנודתי לעומת מצב 0 הדומיננטי. ב־\asset{DIS}, מצב 3 תנודתי מאוד. תיאורים כגון ``נדיר ותנודתי'' נתמכים בטבלה; תוויות כגון ``משבר'', ``שורי'' או ``דובי'' דורשות בדיקה מול אירועים, יציבות בין זרעים ורווחי סמך, ולכן אינן מוקצות כאן.

גם תוויות המספר עצמן אינן מזוהות בין התאמות: \tech{label switching} מאפשר למצב זהה רעיונית לקבל מספר אחר בזרע אחר. הארטיפקט מתעד את כל האתחולים אך אינו כולל התאמת תוויות ומדדי \tech{ARI} ביניהם. לכן אי אפשר לטעון עדיין ליציבות סמנטית של המצבים. עבודות על משטרים מדגישות את הצורך להפריד בין התאמה סטטיסטית לבין סיפור כלכלי \cite{bulla2006application}.

\subsection{משמעות חוצת־נכסים}
היעדר יתרון אינו מוגבל לסוג נכס אחד: הוא מופיע במניות רחבות, זהב, אג"ח, קריפטו ומניה בודדת. מצד שני, המדגם של חמש סדרות קטן ונבחר לצורך גיוון, לא לצורך דגימה סטטיסטית של כל שוקי ההון. \asset{BTC-USD} שונה בלוח המסחר, בתנודתיות ובאיכות אפשרית של נתוני נפח; אין לאחד את כל הנכסים ללא משקלול או מודל היררכי. הממוצע בין הנכסים הוא סיכום תיאורי בלבד.

\subsection{אין מסקנה על רווחיות או מובהקות}
הניסוי אינו אסטרטגיית מסחר. לא הוגדרו פוזיציות, עלויות, החלקת מחיר, מגבלות נזילות, תשואה מצטברת או סיכון. לכן אסור להסיק מן ה־\tech{DPA} רווחיות. בנוסף, לא בוצע ניתוח תקף של תלות סדרתית, רווחי סמך או תיקון לריבוי השוואות. טענות מובהקות היסטוריות שנמצאו בטיוטות ישנות אינן נתמכות בארטיפקט הקנוני ואינן מופיעות כאן.

\section{מגבלות ואיומים על התוקף}
\begin{enumerate}
  \item \textbf{פיצול זמן יחיד:} התוצאה רגישה לתקופת המבחן המסוימת. אין \tech{rolling-origin}, ולכן לא נבדקה יציבות על פני משטרי מאקרו שונים.
  \item \textbf{יקום קטן:} חמישה נכסים אינם מאפשרים הכללה לכל השווקים. \asset{DIS} היא גם מקרה היסטורי מסוים, אף שלא נבחרה לפי הצלחה בריצה הקנונית.
  \item \textbf{חסרי provenance ב־manifest:} תאריכי גבול בפועל, מספר תצפיות לכל מקטע, גרסאות החבילות והאשים של קובצי הקלט אינם נשמרים. השחזור תלוי במקור נתונים חיצוני שעלול להשתנות.
  \item \textbf{אימות אינו מצורף מחדש:} לאחר הבחירה המודל מאומן שוב על האימון בלבד. הדבר מונע בלבול בפרוטוקול אך אינו מנצל את כל המידע שקדם למבחן.
  \item \textbf{בחירה לפי צפיפות ולא לפי יעד החיזוי:} \tech{validation log-likelihood} עשוי להעדיף הפרדת תנודתיות שאינה משפרת סימן תשואה.
  \item \textbf{חשיפה תפעולית ללייקלי-הוד מבחן:} הקוד חישב \tech{test log-likelihood} לכל התחלה לפני הבחירה, אף שהבחירה עצמה אינה משתמשת בו. פרוטוקול קשיח יותר ימנע חישוב כל מדד מבחן עד לנעילה.
  \item \textbf{פענוח נקודתי ולא הסתברותי:} התחזית משתמשת במצב \tech{Viterbi} יחיד; אי־ודאות פוסטריורית אינה משוקללת. סביב גבולות מצב הדבר עלול להיות לא יציב.
  \item \textbf{הנחת גאוסיאן:} תשואות פיננסיות עשויות להיות כבדות־זנב וא־סימטריות. ריבוי מצבים עלול לשמש תחליף גס להתפלגות פליטה מתאימה יותר.
  \item \textbf{בסיס ``שרשרת מרקוב'':} המימוש הוא ממוצע מותנה בכיוון ולא מטריצת מעבר מפורשת; יש לתקן את השם או את המימוש.
  \item \textbf{רצף הקשר:} חישוב הממוצע הנע והערכת המודל אינם משתמשים במקטע האימות כהיסטוריה לאחר הבחירה. בנוסף, גרפי הריצה מחברים אימון ומבחן ומדלגים על האימות, ולכן הם איורי המחשה ולא ציר זמן מלא.
  \item \textbf{זמן ריצה לא אחיד:} זמני הבסיס וה־\tech{HMM} אינם כוללים בדיוק אותם שלבים, ולכן אין להסיק מהם יחס האצה מדויק.
  \item \textbf{היעדר הסקה סטטיסטית:} ללא תחזיות יום־יום שמורות לכל מודל, \tech{block bootstrap} או מבחן זוגי תלוי־זמן לא ניתן לכמת אי־ודאות בפערי \tech{DPA}.
\end{enumerate}

\section{מסקנות}
העבודה בדקה באופן קנוני וזהיר אם \tech{Gaussian HMM} עם מצבים סמויים משפר חיזוי כיוון יומי. הפרוטוקול תיקן זליגת מטרה בגבולות, שמר על סדר כרונולוגי, התאים נרמול באימון בלבד, בדק $K=2,3,4$ בשלושה אתחולים ובחר לפי אימות לפני הערכת מבחן.

התוצאה המרכזית ברורה: ה־\tech{HMM} \textbf{לא ניצח} את מודל ממוצע האימון ב־\tech{DPA}. הממוצע בין הנכסים היה \pct{52.81} לעומת \pct{52.99}; נרשמו ארבעה שוויונות והפסד אחד. לפיכך אין בסיס לטעון לעליונות בחיזוי, למובהקות או לרווחיות. במקביל, המודל זיהה מצבים בעלי שכיחויות, תנודויות ומשכי שהייה שונים, ובכך סיפק ייצוג תיאורי מעניין של הטרוגניות בשוק.

הניסוי הבא המתבקש הוא \tech{rolling-origin} שנקבע מראש, עם מספר חלונות מבחן לא־חופפים, התאמת תוויות בין זרעים, שמירת תחזיות ברמת היום ו־\tech{block bootstrap} לפערים מול בסיסים. ראוי גם להשוות פליטות כבדות־זנב או מודל \tech{Hidden Semi-Markov}, ולבחון בחירה לפי יעד חיזוי באימות. עד אז, המסקנה המדעית היא צנועה: זיהוי משטרים סטטיסטיים אינו מבטיח יתרון בחיזוי הכיוון של היום הבא.
